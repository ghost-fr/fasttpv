using System.Diagnostics;
using System.Text;
using FastTPV.Core.Models;
using Serilog;

namespace FastTPV.Desktop.Features;

/// <summary>
/// Writes a plain-text receipt to disk and, on Windows, attempts to send it to the
/// system print queue via the shell "print" verb. That verb doesn't exist on
/// Linux/macOS, so printing is best-effort there — the text file is always written
/// and returned regardless of whether the print attempt succeeds.
/// </summary>
public sealed class ReceiptService
{
    private static readonly ILogger Logger = Log.ForContext<ReceiptService>();

    public async Task<string> CreateAndPrintAsync(Sale sale, string printerName, string format)
    {
        var directory = Path.Combine(
            Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "FastTPV", "Receipts");
        Directory.CreateDirectory(directory);

        var path = Path.Combine(directory, $"{sale.TicketNumber}-{DateTime.UtcNow:yyyyMMddHHmmss}.txt");
        var width = format == "58mm" ? 32 : 48;

        var receipt = new StringBuilder()
            .AppendLine("FASTTPV")
            .AppendLine(new string('-', width))
            .AppendLine($"Ticket: {sale.TicketNumber}")
            .AppendLine(sale.SaleDate.ToString("g"));

        foreach (var item in sale.LineItems)
            receipt.AppendLine($"{item.Quantity,3} {item.ArticleName,-24} {item.LineTotal,9:0.00}");

        receipt
            .AppendLine(new string('-', width))
            .AppendLine($"TOTAL: {sale.TotalAmount:0.00}")
            .AppendLine(sale.PaymentMethod);

        await File.WriteAllTextAsync(path, receipt.ToString());

        if (!string.IsNullOrWhiteSpace(printerName) &&
            !printerName.Equals("Default Printer", StringComparison.OrdinalIgnoreCase) &&
            OperatingSystem.IsWindows())
        {
            try
            {
                Process.Start(new ProcessStartInfo { FileName = path, Verb = "print", UseShellExecute = true });
            }
            catch (Exception ex)
            {
                // Printing is best-effort — the receipt file itself is always saved,
                // so a failed print attempt shouldn't interrupt checkout.
                Logger.Warning(ex, "Could not send receipt {Path} to printer {Printer}", path, printerName);
            }
        }

        return path;
    }
}
